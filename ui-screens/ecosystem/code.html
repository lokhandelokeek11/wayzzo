// 2. Execute instant proximity route

  origin: { lat: 12.9716, lng: 77.5946 },
  destination: { lat: 12.9352, lng: 77.6245 },
  priority: 'DETERMINISTIC_EXPRESS',
  timeoutMs: 120
});

// 3. Ingestion acknowledge
